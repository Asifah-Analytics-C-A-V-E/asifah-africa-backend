#  Pinned dependencies for Africa backend. Matches the canonical
#  set used by ME (asifah-backend) wherever possible. The ME
#  backend is the canonical source for shared deps; Africa-only
#  divergences are commented inline.
#
#  WHEN ADDING NEW DEPS: pin the version. Unpinned deps cause
#  reproducibility issues across Render rebuilds.
# ════════════════════════════════════════════════════════════════

# ── Core web framework ────────────────────────────────────────
Flask==3.0.0
Flask-CORS==4.0.0
gunicorn==21.2.0

# ── HTTP / networking ─────────────────────────────────────────
# Werkzeug + urllib3 are intentionally NOT pinned — let Flask and
# requests pull the correct transitive versions. Matches ME pattern.
requests==2.31.0

# ── RSS / HTML / XML parsing ──────────────────────────────────
# feedparser handles RSS 2.0 + Atom 1.0; beautifulsoup4 + lxml are
# the canonical fallback when feedparser chokes on malformed feeds
# (which African source outlets sometimes produce).
feedparser==6.0.11
beautifulsoup4==4.12.3
lxml==5.1.0

# ── Redis (Python protocol) ───────────────────────────────────
# Our canonical pattern uses Upstash REST API via the `requests`
# library, so `redis-py` isn't strictly required — but we pin it
# for parity with ME in case any future tracker uses redis-py
# directly (e.g., pub/sub, pipelining).
redis==5.0.1

# ── Telegram OSINT ingestion ──────────────────────────────────
# Telethon is the Telegram MTProto client. We deliberately do NOT
# pin cryptg (the Rust-accelerated crypto helper) — it requires a
# Rust toolchain + writable cache during build, which Render's
# build environment doesn't provide. Telethon falls back to a
# pure-Python crypto path when cryptg isn't available; ~30%
# slower per message but functionally identical. The May 24 2026
# deploy attempt failed on cryptg's Rust build step; removing it
# resolved the deploy.
telethon==1.36.0

# ── Date / time handling ──────────────────────────────────────
python-dateutil==2.8.2

# ── DELIBERATELY OMITTED ──────────────────────────────────────
# yfinance — ME hosts the canonical commodity tracker; Africa
#            consumes commodity data via /api/commodity-pressure
#            proxy. No need for yfinance here.
